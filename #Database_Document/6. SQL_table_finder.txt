# 오라클 DB에서 내 계정에 들어간 진짜 테이블 이름 '철자' 확인하기
# 툴이 인식하고 있는 정확한 대소문자와 철자를 딕셔너리 뷰에서 긁어오기 
# 아래 쿼리를 그대로 복사해서 실행

SELECT table_name FROM user_tables WHERE table_name LIKE '%Jipdaum%';

# 출력된 결과가 대문자(Jipdaum_PRODUCT)라면 : SELECT * FROM Jipdaum_PRODUCT; (따옴표 없음)
# 출력된 결과가 소문자(Jipdaum_product)라면 : SELECT * FROM "Jipdaum_product"; (반드시 소문자 + 쌍따옴표)
